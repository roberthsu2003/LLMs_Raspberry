# 保存會話到檔案

> 💾 很有存在感

```python
"""
title: Export Conversation
author: Your Name
version: 0.1.0
"""

from pydantic import BaseModel
from datetime import datetime
import os

class Action:
    class Valves(BaseModel):
        export_format: str = "markdown"  # or "txt", "json"

    def __init__(self):
        self.valves = self.Valves()

    async def action(
        self,
        body: dict,
        __user__=None,
        __event_emitter__=None,
        __event_call__=None,
    ) -> Optional[dict]:
        """匯出對話到本機"""
        
        conversation_content = body.get("message", "")
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        
        if __event_emitter__:
            await __event_emitter__({
                "type": "status",
                "data": {
                    "description": f"正在保存為 {self.valves.export_format}...",
                    "done": False
                }
            })
            
            # 組織檔案名稱
            filename = f"conversation_{timestamp}.{self.valves.export_format}"
            filepath = f"/app/data/{filename}"
            
            try:
                # 確保目錄存在（Docker 中）
                os.makedirs("/app/data", exist_ok=True)
                
                with open(filepath, "w", encoding="utf-8") as f:
                    f.write(conversation_content)
                
                await __event_emitter__({
                    "type": "message",
                    "data": {
                        "content": f"✅ 已保存！\n檔案名: {filename}\n路徑: {filepath}"
                    }
                })
                
                await __event_emitter__({
                    "type": "status",
                    "data": {
                        "description": "導出完成",
                        "done": True
                    }
                })
            except Exception as e:
                await __event_emitter__({
                    "type": "message",
                    "data": {
                        "content": f"❌ 錯誤: {str(e)}"
                    }
                })
        
        return {"status": "exported"}
```
