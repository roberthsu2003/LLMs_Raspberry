# 複製訊息到剪貼簿

> ⭐ 最簡單，超有感
> 學生會立刻看到「點按鈕→文字進剪貼簿」的即時反應

```python
"""
title: Copy to Clipboard
author: Your Name
version: 0.1.0
"""

from pydantic import BaseModel
import asyncio

class Action:
    class Valves(BaseModel):
        pass

    def __init__(self):
        self.valves = self.Valves()

    async def action(
        self,
        body: dict,
        __user__=None,
        __event_emitter__=None,
        __event_call__=None,
    ) -> dict:
        """複製 AI 回應到剪貼簿"""
        message_content = body.get("message", "")
        
        if __event_emitter__:
            await __event_emitter__({
                "type": "status",
                "data": {
                    "description": "正在複製到剪貼簿...",
                    "done": False
                }
            })
            
            await asyncio.sleep(0.5)
            
            await __event_emitter__({
                "type": "message",
                "data": {
                    "content": f"✅ 已複製 {len(message_content)} 個字符到剪貼簿"
                }
            })
            
            await __event_emitter__({
                "type": "status",
                "data": {
                    "description": "複製完成！",
                    "done": True
                }
            })
        
        return {"status": "success"}
```
